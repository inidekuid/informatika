💻 Kode yang sedang dijalankan

class AplikasiKetik:
def __init__(self):
self.teks = ""
self.riwayat = [] # STACK
 
def ketik_kata(self, kata_baru):
self.riwayat.append(self.teks) # PUSH ke stack
self.teks += kata_baru
print(f"Hasil: '{self.teks}'")
 
def tekan_undo(self):
if len(self.riwayat) > 0:
self.teks = self.riwayat.pop() # POP dari stack
print(f"[UNDO] Teks: '{self.teks}'")
else:
print("Tidak ada yang bisa di-undo!")
